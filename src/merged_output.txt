#include "backlight.h"

Backlight* Backlight::_self = nullptr;

void Backlight::Use(Backlight& instance) {
  _self = &instance;
}

void Backlight::beginStatic() {
  if (_self) _self->begin();
}
void Backlight::nextStatic() {
  if (_self) _self->next();
}
uint8_t Backlight::currentPercentStatic() {
  return _self ? _self->currentPercent() : 100;
}
void Backlight::setPercentStatic(uint8_t p) {
  if (_self) _self->setPercent(p);
}

void Backlight::setLevels(const uint8_t* levels, size_t count) {
  _levels = levels;
  _count = count;
  _idx = 0;
}

uint8_t Backlight::currentPercent() const {
  if (!_levels || _count == 0) return 100;
  return _levels[_idx];
}

void Backlight::begin() {
  BacklightConfig cfg;
  begin(cfg);
}

void Backlight::begin(const BacklightConfig& cfg) {
  _cfg = cfg;
  _maxDuty = (1u << _cfg.resBits) - 1;

  static const uint8_t defaultLevels[] = {100, 70, 30, 10, 1};
  if (!_levels || _count == 0) {
    setLevels(defaultLevels, sizeof(defaultLevels) / sizeof(defaultLevels[0]));
  }

  pinMode(_cfg.pin, OUTPUT);

  backlightInit();
  setPercent(currentPercent());
}

void Backlight::backlightInit() {
  ledcSetup(_cfg.channel, _cfg.freq, _cfg.resBits);
  ledcAttachPin(_cfg.pin, _cfg.channel);
}

void Backlight::backlightForceMax() {
  ledcWrite(_cfg.channel, 0);
  delay(1);

  ledcDetachPin(_cfg.pin);
  delay(2);

  pinMode(_cfg.pin, OUTPUT);
  digitalWrite(_cfg.pin, HIGH);
  delay(2);

  Serial.printf("FORCE MAX: gpio=%d\n", digitalRead(_cfg.pin));
}

void Backlight::setPercent(uint8_t percent) {
  if (percent >= 100) {
    backlightForceMax();
    return;
  }
  if (percent < 1) percent = 1;

  ledcAttachPin(_cfg.pin, _cfg.channel);
  uint32_t duty = (uint32_t)percent * _maxDuty / 100;
  ledcWrite(_cfg.channel, duty);

  Serial.printf("BL %u%% duty=%u/%u\n", percent, duty, _maxDuty);
}

void Backlight::next() {
  if (!_levels || _count == 0) return;
  _idx = (_idx + 1) % _count;
  uint8_t p = _levels[_idx];
  Serial.printf("Brightness step: %u%%\n", p);
  setPercent(p);
}

#include "buttons.h"

// ---------- Button ----------

void Button::begin(uint8_t pin, bool activeLow, bool usePullup) {
  _s.pin = pin;
  _s.activeLow = activeLow;

  if (usePullup && activeLow) {
    pinMode(pin, INPUT_PULLUP);
  } else {
    pinMode(pin, INPUT);
  }

  _s.lastLevel = digitalRead(pin);
  _s.lastChangeMs = millis();

  _s.pressed = false;
  _s.pressStartMs = 0;
  _s.longFired = false;
  _s.nextRepeatMs = 0;

  _s.clickPending = false;
  _s.clickDeadlineMs = 0;
}

void Button::fireSingleIfDue(uint32_t now) {
  if (_s.clickPending && now >= _s.clickDeadlineMs) {
    _s.clickPending = false;
    if (onSingle) onSingle();
  }
}

void Button::handle(uint32_t now) {
  if (_s.pin == 255) return;

  bool level = digitalRead(_s.pin);

  // debounce
  if (level != _s.lastLevel) {
    _s.lastLevel = level;
    _s.lastChangeMs = now;
    return;
  }
  if (now - _s.lastChangeMs < debounceMs) return;

  fireSingleIfDue(now);

  bool down = isPressedLevel(level);

  // press
  if (down && !_s.pressed) {
    _s.pressed = true;
    _s.pressStartMs = now;
    _s.longFired = false;
    _s.nextRepeatMs = onRepeat ? (now + repeatDelayMs) : 0;
    return;
  }

  // hold
  if (down && _s.pressed) {
    if (!_s.longFired && onLong && now - _s.pressStartMs >= longPressMs) {
      _s.longFired = true;
      onLong();
    }
    if (onRepeat && _s.nextRepeatMs && now >= _s.nextRepeatMs) {
      onRepeat();
      _s.nextRepeatMs = now + repeatEveryMs;
    }
    return;
  }

  // release
  if (!down && _s.pressed) {
    _s.pressed = false;
    _s.nextRepeatMs = 0;

    if (_s.longFired) {
      _s.longFired = false;
      _s.clickPending = false;
      return;
    }

    if (!onDouble || doubleClickMs == 0) {
      if (onSingle) onSingle();
      return;
    }

    if (!_s.clickPending) {
      _s.clickPending = true;
      _s.clickDeadlineMs = now + doubleClickMs;
    } else {
      _s.clickPending = false;
      if (onDouble) onDouble();
    }
  }
}

void Button::updateAt(uint32_t now) {
  handle(now);
  fireSingleIfDue(now);
}

void Button::update() {
  updateAt(millis());
}

namespace Buttons {

static constexpr size_t MAX_BTNS = 8;
static Button* g_btns[MAX_BTNS] = {nullptr};

#if defined(ARDUINO_ARCH_ESP32)
static TaskHandle_t g_task = nullptr;
static constexpr uint32_t POLL_MS     = 20;
static constexpr int      CORE        = 1;
static constexpr uint32_t STACK_WORDS = 2048;
static constexpr UBaseType_t PRIO     = 2;

static void tickAll() {
  uint32_t now = millis();
  for (auto& b : g_btns) {
    if (b) b->updateAt(now);
  }
}

static void taskFn(void*) {
  TickType_t last = xTaskGetTickCount();
  for (;;) {
    tickAll();
    vTaskDelayUntil(&last, pdMS_TO_TICKS(POLL_MS));
  }
}
#endif

bool attach(Button* b) {
  if (!b) return false;
  for (auto& slot : g_btns) {
    if (slot == b) return true;
  }
  for (auto& slot : g_btns) {
    if (!slot) {
      slot = b;
      return true;
    }
  }
  return false;
}

bool detach(Button* b) {
  for (auto& slot : g_btns) {
    if (slot == b) {
      slot = nullptr;
      return true;
    }
  }
  return false;
}

void beginTask() {
#if defined(ARDUINO_ARCH_ESP32)
  if (g_task) return;
  xTaskCreatePinnedToCore(
    taskFn,
    "buttons",
    STACK_WORDS,
    nullptr,
    PRIO,
    &g_task,
    CORE
  );
#endif
}

void stopTask() {
#if defined(ARDUINO_ARCH_ESP32)
  if (g_task) {
    vTaskDelete(g_task);
    g_task = nullptr;
  }
#endif
}

bool isTaskRunning() {
#if defined(ARDUINO_ARCH_ESP32)
  return g_task != nullptr;
#else
  return false;
#endif
}

} // namespace Buttons

    bool     clickPending = false;

// logs.h
#pragma once
#include <Arduino.h>
#include <stdarg.h>

enum class LogLevel : uint8_t { Info, Warn, Error };

template <size_t CAP, size_t MSG_LEN>
class LogBufferT {
public:
  struct Entry {
    uint32_t ms;
    LogLevel level;
    char msg[MSG_LEN];
  };

  void add(LogLevel lvl, const char* s) {
    Entry& e = _buf[_head];
    e.ms = millis();
    e.level = lvl;
    strlcpy(e.msg, s ? s : "", MSG_LEN);
    _head = (_head + 1) % CAP;
    if (_count < CAP) _count++;
  }

  void add(LogLevel lvl, const String& s) {
    add(lvl, s.c_str());
  }

  void addf(LogLevel lvl, const char* fmt, ...) {
    char tmp[MSG_LEN];
    va_list ap;
    va_start(ap, fmt);
    vsnprintf(tmp, sizeof(tmp), fmt, ap);
    va_end(ap);
    add(lvl, tmp);
  }

  size_t count() const { return _count; }

  Entry getNewestFirst(size_t i) const {
    // i=0 самый новый
    if (i >= _count) return Entry{0, LogLevel::Info, ""};
    int idx = (int)_head - 1 - (int)i;
    while (idx < 0) idx += CAP;
    return _buf[idx];
  }

private:
  Entry _buf[CAP]{};
  size_t _head = 0;
  size_t _count = 0;
};

  }

#include "nightscout_client.h"

  _cfg = cfg;

String NightscoutClient::buildBase(String url) const {
  url.trim();
  if (url.isEmpty()) return url;

  if (!url.startsWith("http://") && !url.startsWith("https://")) {
    url = String(_cfg.preferHttpsIfNoScheme ? "https://" : "http://") + url;
  }
  if (url.endsWith("/")) url.remove(url.length() - 1);
  return url;
}

String NightscoutClient::makeEntriesUrl(const String& base) const {
  String u = base;
  if (_cfg.sgvOnly) {
    u += "/api/v1/entries.json?find[type][$eq]=sgv&count=10";
  } else {
    u += "/api/v1/entries.json?count=10";
  }

  if (!_cfg.token.isEmpty()) {
    // entries всегда уже с "?"
    u += "&token=";
    u += _cfg.token;
  }
  return u;
}

String NightscoutClient::makePropsUrl(const String& base) const {
  String u = base;

  if (_cfg.infoLine == 2) {
    u += "/api/v2/properties/iob,cob,delta,loop,basal";
  } else if (_cfg.infoLine == 3) {
    u += "/api/v2/properties/iob,cob,delta,openaps,basal";
  } else {
    u += "/api/v2/properties/iob,cob,delta,basal";
  }

  if (!_cfg.token.isEmpty()) {
    // props без query до токена
    u += "?token=";
    u += _cfg.token;
  }
  return u;
}

void NightscoutClient::sanitizeJson(String& s) const {
  // как у вас: контрольные символы -> пробел
  for (int i = 0; i < (int)s.length(); i++) {
    if ((uint8_t)s[i] < 32) s.setCharAt(i, ' ');
  }
  s.replace("\\u0000", " ");
  s.replace("\\u000b", " ");
  s.replace("\\u0032", " ");

  // ваша “защита date: ... .123”
  int ndx = 0;
  int sr = s.indexOf("\"date\":", ndx);
  while (sr != -1) {
    ndx = sr + 1;
    if (sr + 20 < (int)s.length()) {
      if (s.charAt(sr + 20) == '.') {
        s.remove(sr + 20, 1);
        while (sr + 20 < (int)s.length()) {
          char c = s.charAt(sr + 20);
          if (c >= '0' && c <= '9') s.remove(sr + 20, 1);
          else break;
        }
      }
    }
    sr = s.indexOf("\"date\":", ndx);
  }
}

bool NightscoutClient::httpGet(const String& url, String& body, int& httpCode) const {
  HTTPClient http;
  http.setTimeout(_cfg.httpTimeoutMs);

  Serial.printf("[NS] GET %s\n", url.c_str());

  // Важно: клиент должен жить ДО КОНЦА функции, а не внутри if-блока
  WiFiClientSecure clientSecure;
  WiFiClient       clientPlain;

  bool begun = false;
  if (url.startsWith("https://")) {
    clientSecure.setInsecure();                 // если нужен реальный TLS — поставьте CA вместо setInsecure()
    begun = http.begin(clientSecure, url);
  } else {
    begun = http.begin(clientPlain, url);
  }

  if (!begun) {
    Serial.println("[NS] http.begin FAILED");
    httpCode = -1;
    body = "";
    return false;
  }

  uint32_t t0 = millis();
  httpCode = http.GET();
  uint32_t dt = millis() - t0;

  // Печатаем код один раз, единообразно
  Serial.printf("[NS] code=%d, %lu ms\n", httpCode, (unsigned long)dt);

  // Читаем body всегда (даже при 400/401/500) — это сильно помогает диагностике
  body = http.getString();
  Serial.printf("[NS] bodyLen=%u\n", (unsigned)body.length());
  if (body.length() && body.length() < 512) {   // чтобы не заспамить Serial
    Serial.printf("[NS] body: %s\n", body.c_str());
  } else if (body.length() >= 512) {
    Serial.println("[NS] body: <too long, skipped>");
  }

  http.end();

  // Успех — только 200 OK (при желании можно разрешить 204 и т.п.)
  if (httpCode == HTTP_CODE_OK) return true;

  // Пояснение ошибки (для отрицательных кодов errorToString часто пустой)
  if (httpCode < 0) {
    Serial.printf("[NS] transport error: %s\n", http.errorToString(httpCode).c_str());
  } else {
    Serial.println("[NS] HTTP error");
  }
  return false;
}

bool NightscoutClient::parseEntries(const String& json, NightscoutData& out, ArduinoJson::DynamicJsonDocument& doc) const {
  DeserializationError e = deserializeJson(doc, json);
  if (e) {
    NS_LOGE("[NS] entries JSON parse failed: %s", e.c_str());
    return false;
  }

  JsonArray arr = doc.as<JsonArray>();
  if (arr.isNull() || arr.size() == 0) {
    NS_LOGE("[NS] entries empty");
    return false;
  }

  // найти первый объект с "sgv" (как у вас)
  int sgvindex = 0;
  JsonObject obj;
  do {
    obj = doc[sgvindex].as<JsonObject>();
    sgvindex++;
  } while (!obj.containsKey("sgv") && (sgvindex < (int)arr.size() - 1));
  sgvindex--;

  if (sgvindex < 0 || sgvindex >= (int)arr.size()) sgvindex = 0;

  strlcpy(out.dev, doc[sgvindex]["device"] | "N/A", sizeof(out.dev));
  strlcpy(out.dir, doc[sgvindex]["direction"] | "N/A", sizeof(out.dir));
  if (strcmp(out.dir, "N/A") == 0) {
    strlcpy(out.dir, doc[sgvindex]["trend"] | "N/A", sizeof(out.dir));
  }

  out.rawtime_ms = doc[sgvindex]["date"].as<long long>();
  out.sensTime   = (time_t)(out.rawtime_ms / 1000LL);

  out.sgv_mgdl = doc[sgvindex]["sgv"] | 0.0f;
  out.sgv_mmol = mgdlToMmol(out.sgv_mgdl);

  for (int i = 0; i < 10; i++) {
    float v = doc[i]["sgv"] | 0.0f;
    out.last10mmol[i] = mgdlToMmol(v);
  }

  NS_LOGI("[NS] entries OK dev='%s' dir='%s' sgv=%.1f mmol (%.0f mg/dL)",
          out.dev, out.dir, out.sgv_mmol, out.sgv_mgdl);

  return true;
}

bool NightscoutClient::parseProps(const String& json, NightscoutData& out, ArduinoJson::DynamicJsonDocument& doc) const {
  DeserializationError e = deserializeJson(doc, json);
  if (e) {
    NS_LOGE("[NS] props JSON parse failed: %s", e.c_str());
    return false;
  }

  JsonObject iob = doc["iob"];
  out.iob = iob["iob"] | 0.0f;

  JsonObject cob = doc["cob"];
  out.cob = cob["cob"] | 0.0f;

  JsonObject delta = doc["delta"];
  out.delta_mgdl = delta["mgdl"] | 0;
  out.delta_mmol = mgdlToMmol((float)out.delta_mgdl);

  if (_cfg.showMgdl) {
    snprintf(out.delta_display, sizeof(out.delta_display), "%+d", out.delta_mgdl);
  } else {
    snprintf(out.delta_display, sizeof(out.delta_display), "%+.1f", out.delta_mmol);
  }

  JsonObject loop_obj;
  JsonObject loop_display;
  if (_cfg.infoLine == 3) {
    loop_obj = doc["openaps"];
    loop_display = loop_obj["status"];
  } else {
    loop_obj = doc["loop"];
    loop_display = loop_obj["display"];
  }

  strlcpy(out.loop_code,  loop_display["code"]  | "N/A", sizeof(out.loop_code));
  strlcpy(out.loop_label, loop_display["label"] | "N/A", sizeof(out.loop_label));

  JsonObject basal = doc["basal"];
  strlcpy(out.basal_display, basal["display"] | "N/A", sizeof(out.basal_display));

  NS_LOGI("[NS] props OK iob=%.2f cob=%.0f delta=%s loop=%s basal=%s",
          out.iob, out.cob, out.delta_display, out.loop_label, out.basal_display);

  return true;
}

bool NightscoutClient::update(NightscoutData& out) {
  out = NightscoutData{}; // как “обнуление” в вашем readNightscout()

  if (WiFi.status() != WL_CONNECTED) {
    NS_LOGE("[NS] WiFi not connected");
    return false;
  }

  String base = buildBase(_cfg.baseUrl);
  if (base.isEmpty()) {
    NS_LOGE("[NS] baseUrl empty");
    return false;
  }

  // общий doc как в проекте (16384 у вас часто хватает)
  ArduinoJson::DynamicJsonDocument doc(16384);

  // ----- entries -----
  {
    const String url = makeEntriesUrl(base);
    String json;
    int httpCode = 0;

    if (httpGet(url, json, httpCode)) {
      out.httpEntries = httpCode;
      sanitizeJson(json);
      out.okEntries = parseEntries(json, out, doc);
    } else {
      out.httpEntries = httpCode;
      out.okEntries = false;
    }
  }

  // ----- props -----
  {
    const String url = makePropsUrl(base);
    String json;
    int httpCode = 0;

    if (httpGet(url, json, httpCode)) {
      out.httpProps = httpCode;
      sanitizeJson(json);
      out.okProps = parseProps(json, out, doc);
    } else {
      out.httpProps = httpCode;
      out.okProps = false;
    }
  }

  return out.okEntries || out.okProps;
}

static int drawWrappedLine(TFT_eSprite& sprite, const String& text, int x, int y,
                           int maxW, int lineH, int maxLines)
{
  int linesUsed = 0;
  int start = 0;
  const int n = (int)text.length();

  while (start < n && linesUsed < maxLines) {
    // пропустить ведущие пробелы
    while (start < n && text[start] == ' ') start++;
    if (start >= n) break;

    int end = start;
    int lastSpace = -1;

    // расширяем до тех пор, пока влезает
    while (end < n) {
      char c = text[end];
      if (c == ' ') lastSpace = end;

      String candidate = text.substring(start, end + 1);
      if (sprite.textWidth(candidate) > maxW) break;
      end++;
    }

    int cut;
    if (end == start) {
      // даже 1 символ не влез — аварийный сдвиг
      cut = start + 1;
    } else if (end >= n) {
      // всё влезло
      cut = n;
    } else {
      // не влезло на end
      if (lastSpace >= start) {
        cut = lastSpace; // перенос по пробелу
      } else {
        cut = end;       // перенос внутри слова
      }
    }

    String line = text.substring(start, cut);
    line.trim(); // убрать пробелы по краям
    sprite.drawString(line, x, y + linesUsed * lineH);

    linesUsed++;

    // следующий кусок: после пробела (если перенос по пробелу)
    start = cut;
    while (start < n && text[start] == ' ') start++;
  }

  return linesUsed;
}

void drawLogScreen(TFT_eSprite& sprite, const LogBuffer& logs) {
  const uint16_t bg = TFT_BLACK;
  sprite.fillSprite(bg);

  // Максимально мелко и плотно
  sprite.setTextFont(1);
  sprite.setTextSize(1);
  sprite.setTextDatum(TL_DATUM);

  const int lineH = sprite.fontHeight(); // максимально плотный шаг
  const int x = 0;
  int y = 0;

  const int screenW = 320;
  const int screenH = 170;

  int linesLeft = screenH / lineH;       // сколько строк всего влезает

  // Печатаем самые новые сверху, но с переносом каждая запись может занять несколько строк
  for (int i = 0; i < (int)logs.count() && linesLeft > 0; i++) {
    LogEntry e = logs.getNewestFirst(i);

    uint16_t col = TFT_GREEN;
    if (e.level == LogLevel::Warn)  col = TFT_YELLOW;
    if (e.level == LogLevel::Error) col = TFT_RED;   // только ошибки красным

    sprite.setTextColor(col, bg);

    //char tag = (e.level == LogLevel::Info) ? 'I' : (e.level == LogLevel::Warn) ? 'W' : 'E';
    uint32_t sec = e.ms / 1000;

    //String msg = String(sec) + " " + String(tag) + " " + e.msg;
    String msg = String(sec) + " "  + e.msg;

    // рисуем с переносом
    int used = drawWrappedLine(sprite, msg, x, y, screenW, lineH, linesLeft);
    y += used * lineH;
    linesLeft -= used;
  }

  sprite.pushSprite(0, 0);
}

#include "time_utils.h"

// timezone
static constexpr int timezone = 5; // как у вас
static constexpr int timezoneOffset = timezone * SECS_PER_MIN;

bool isDST(time_t timestamp) {
  if (month(timestamp) < 3 || month(timestamp) > 11) return false;

  int secondSunInMarch = 14 - ((5 * year(timestamp) / 4 + 1) % 7);
  int firstSunInNovember = 7 - ((5 * year(timestamp) / 4 + 1) % 7);

  if (month(timestamp) == 3) {
    if (day(timestamp) > secondSunInMarch) return true;
    else if (day(timestamp) == secondSunInMarch && hour(timestamp) >= 2) return true;
  }

  if (month(timestamp) == 11) {
    if (day(timestamp) < firstSunInNovember) return true;
    else if (day(timestamp) == firstSunInNovember && hour(timestamp) < 2) return true;
  }
  return false;
}

void adjustTimezone(time_t& timestamp) {
  timestamp += timezoneOffset * SECS_PER_MIN;
  if (isDST(timestamp)) timestamp += SECS_PER_HOUR;
}

void WifiManager::addNetworks(const WifiCred* list, size_t count) {
  for (size_t i = 0; i < count; i++) {
    if (!list[i].ssid || list[i].ssid[0] == '\0') continue;

    const char* pass = list[i].pass;
    if (!pass || pass[0] == '\0') {
      logI(String("[WiFi] addAP: '") + list[i].ssid + "' (open)");
      _multi.addAP(list[i].ssid, nullptr);
    } else {
      logI(String("[WiFi] addAP: '") + list[i].ssid + "' (password set)");
      _multi.addAP(list[i].ssid, pass);
    }
  }
}

bool WifiManager::begin() {
  _apMode = false;

  logI("[WiFi] ---- begin ----");
  logI(String("[WiFi] hostname: ") + (_cfg.hostname ? _cfg.hostname : "(null)"));
  logI(String("[WiFi] timeout : ") + String((unsigned long)_cfg.connectTimeoutMs) + " ms");

  WiFi.mode(WIFI_STA);
  if (_cfg.hostname && _cfg.hostname[0]) WiFi.setHostname(_cfg.hostname);
  WiFi.setAutoReconnect(_cfg.enableAutoReconnect);

  // очистка предыдущих состояний
  WiFi.disconnect(true, true);
  delay(100);

  uint32_t t0 = millis();

  while ((millis() - t0) < _cfg.connectTimeoutMs) {
    uint8_t st = _multi.run();

    if (st == WL_CONNECTED) {
        logI(String("[WiFi] CONNECTED: SSID='") + WiFi.SSID() + "'");
        logI(String("[WiFi] IP       : ") + WiFi.localIP().toString());
        logI(String("[WiFi] RSSI     : ") + String(WiFi.RSSI()) + " dBm");
        logStatus("[WiFi] status  : ");
        logI("[WiFi] ---- ok ----");
        return true;
    }

    delay(_cfg.retryDelayMs);
  }


  logE("[WiFi] CONNECT TIMEOUT");
  logStatus("[WiFi] status  : ");

  if (_cfg.enableFallbackAP) {
    startFallbackAP();
  } else {
    logW("[WiFi] fallback AP disabled");
  }

  logE("[WiFi] ---- fail ----");
  return false;
}

bool WifiManager::isStaConnected() const {

}

bool WifiManager::isApMode() const {
  return _apMode && (WiFi.getMode() & WIFI_AP);
}

void WifiManager::startFallbackAP() {
  logW("[WiFi] starting fallback AP mode...");

  WiFi.enableAP(true);
  WiFi.mode(WIFI_AP);

  bool cfgOk = WiFi.softAPConfig(_cfg.apIp, _cfg.apGw, _cfg.apMask);
  logI(String("[WiFi] softAPConfig: ") + (cfgOk ? "OK" : "FAIL"));

  bool apOk = WiFi.softAP(_cfg.fallbackApSsid, _cfg.fallbackApPass);
  logI(String("[WiFi] softAP start : ") + (apOk ? "OK" : "FAIL"));

  logI(String("[WiFi] AP SSID      : ") + _cfg.fallbackApSsid);
  logI(String("[WiFi] AP IP        : ") + WiFi.softAPIP().toString());

  _apMode = apOk;

  if (!apOk) {
    logE("[WiFi] AP mode failed");
  }
}

void WifiManager::logStatus(const char* prefix) {
  const char* p = prefix ? prefix : "";

  wl_status_t st = WiFi.status();
  const char* s = "UNKNOWN";
  switch (st) {
    case WL_IDLE_STATUS:      s = "IDLE"; break;
    case WL_NO_SSID_AVAIL:    s = "NO_SSID_AVAIL"; break;
    case WL_SCAN_COMPLETED:   s = "SCAN_COMPLETED"; break;
    case WL_CONNECTED:        s = "CONNECTED"; break;
    case WL_CONNECT_FAILED:   s = "CONNECT_FAILED"; break;
    case WL_CONNECTION_LOST:  s = "CONNECTION_LOST"; break;
    case WL_DISCONNECTED:     s = "DISCONNECTED"; break;
    default:                  s = "UNKNOWN"; break;
  }

  logI(String(p) + s + " (mode=" + String((int)WiFi.getMode()) + ")");
}
